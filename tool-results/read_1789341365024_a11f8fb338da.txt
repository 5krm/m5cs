   159→/** Dark radial blob painted under the car — a fake ambient-occlusion
   160→ *  contact patch. The photo asphalt under the car is often near-black, so
   161→ *  the real cast shadow alone gives no grounding cue; this soft dark
   162→ *  ellipse (which yaws with the car) anchors the wheels to the ground
   163→ *  without reading as a podium (edges fade to fully transparent). */
   164→function makeContactShadowTexture(): THREE.CanvasTexture {
   165→  const canvas = document.createElement('canvas')
   166→  canvas.width = canvas.height = 256
   167→  const ctx = canvas.getContext('2d')!
   168→  const g = ctx.createRadialGradient(128, 128, 10, 128, 128, 128)
   169→  g.addColorStop(0, 'rgba(0,0,0,0.62)')
   170→  g.addColorStop(0.5, 'rgba(0,0,0,0.34)')
   171→  g.addColorStop(0.8, 'rgba(0,0,0,0.1)')
   172→  g.addColorStop(1, 'rgba(0,0,0,0)')
   173→  ctx.fillStyle = g
   174→  ctx.fillRect(0, 0, 256, 256)
   175→  const tex = new THREE.CanvasTexture(canvas)
   176→  tex.colorSpace = THREE.SRGBColorSpace
   177→  return tex
   178→}

   211→type CarRig = {
   212→  car: THREE.Group
   213→}
   214→
   215→/**
   216→ * Normalizes ANY car glTF so the camera keyframes above fit without
   217→ * retuning: centers the footprint, grounds it at y = 0, scales the longest
   218→ * horizontal span to TARGET_LENGTH, repaints body/glass, hides showroom
   219→ * plates and enables shadows.
   220→ */
   221→function buildCarRig(source: THREE.Object3D): CarRig {
   222→  const model = source.clone(true)
   223→
   224→  const box = new THREE.Box3().setFromObject(model)
   225→  const size = box.getSize(new THREE.Vector3())
   226→  const center = box.getCenter(new THREE.Vector3())
   227→  const scale = TARGET_LENGTH / Math.max(size.x, size.z)
   228→  model.scale.setScalar(scale)
   229→  model.position.set(-center.x * scale, -box.min.y * scale, -center.z * scale)
   230→  if (FLIP_MODEL) model.rotation.y = Math.PI
   231→
   232→  const gray = satinGrayPaint()
   233→  const glass = darkGlass()
   234→  const junk: THREE.Object3D[] = []
   235→  model.traverse((obj) => {
   236→    if (!(obj instanceof THREE.Mesh)) return
   237→    obj.castShadow = true
   238→    obj.receiveShadow = false
   239→    const mats = Array.isArray(obj.material) ? [...obj.material] : [obj.material]
   240→    let replaced = false
   241→    for (let i = 0; i < mats.length; i++) {
   242→      const name = mats[i]?.name ?? ''
   243→      if (BODY_PAINT.has(name)) {
   244→        mats[i] = gray
   245→        replaced = true
   246→      } else if (GLASS.has(name)) {
   247→        mats[i] = glass
   248→        replaced = true
   249→      } else if (mats[i] && (mats[i] as THREE.MeshStandardMaterial).isMeshStandardMaterial) {
   250→        ;(mats[i] as THREE.MeshStandardMaterial).envMapIntensity = 0.7
   251→      }
   252→    }
   253→    if (replaced) obj.material = Array.isArray(obj.material) ? mats : mats[0]
   254→    // hide giant flat plates (Sketchfab showroom floors)
   255→    const b = new THREE.Box3().setFromObject(obj)
   256→    if (b.max.y - b.min.y < 0.12 && b.max.x - b.min.x > 3 && b.max.z - b.min.z > 3) junk.push(obj)
   257→  })
   258→  junk.forEach((m) => (m.visible = false))
   259→
   260→  const car = new THREE.Group()
   261→  car.add(model)
   262→
   263→  return { car }
   264→}

   270→type FlatKey = { px: number; py: number; pz: number; tx: number; ty: number; tz: number; fo: number }
   271→
   272→/** Flatten a keyframe; on portrait screens the pos→target offset is
   273→ *  scaled by mobileF so the car never clips out of the narrow viewport.
   274→ *  `fo` resolves the per-shot lens — the ultra-wide `fov` values are a
   275→ *  DESKTOP concern (desktop default 45° is narrow); portrait already uses
   276→ *  the wide FOV_MOBILE, so only an explicit `fovMobile` overrides it. */
   277→function flattenKey(k: CamKey, mobile: boolean): FlatKey {
   278→  let [px, py, pz] = k.pos
   279→  const [tx, ty, tz] = k.target
   280→  if (mobile) {
   281→    px = tx + (px - tx) * k.mobileF
   282→    py = ty + (py - ty) * k.mobileF
   283→    pz = tz + (pz - tz) * k.mobileF * (k.mobileHeadOn ?? 1)
   284→  }
   285→  const fo = mobile ? (k.fovMobile ?? FOV_MOBILE) : (k.fov ?? FOV_DESKTOP)
   286→  return { px, py, pz, tx, ty, tz, fo }
   287→}

   328→export default function ScrollExperience() {
   329→  const canvasRef = useRef<HTMLCanvasElement>(null)

   337→    const canvas = canvasRef.current
   338→    const track = trackRef.current
   339→    const heroLayer = heroLayerRef.current
   340→    const capFront = capFrontRef.current
   341→    const capRear = capRearRef.current
   342→    const endCard = endCardRef.current

   346→    gsap.registerPlugin(ScrollTrigger)
   347→
   348→    const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   349→
   350→    /* ── Renderer ──────────────────────────────────────────────────── */
   351→    const renderer = new THREE.WebGLRenderer({
   352→      canvas,
   353→      antialias: true,
   354→      alpha: false,
   355→      powerPreference: 'high-performance',
   356→    })
   357→    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)) // perf cap
   358→    renderer.setSize(window.innerWidth, window.innerHeight, false)
   359→    renderer.shadowMap.enabled = true
   360→    // three r186 removed PCFSoftShadowMap — PCFShadowMap is the supported
   361→    // filtered shadow path (renders soft contacts with a 2K map)
   362→    renderer.shadowMap.type = THREE.PCFShadowMap
   363→    renderer.toneMapping = THREE.ACESFilmicToneMapping
   364→    renderer.toneMappingExposure = 1.15 // lifts the night IBL without clipping the city lights
   365→    renderer.outputColorSpace = THREE.SRGBColorSpace
   366→
   367→    /* ── Scene: lakeside night city — 360° equirect panorama ─────
   368→     * The SAME texture is the visible backdrop and the IBL source
   369→     * (PMREM pre-filter), so the metallic paint reflects exactly the
   370→     * skyline / lake / bridges you see behind the car. */
   371→    const scene = new THREE.Scene()
   372→    renderer.setClearColor(0x04050a, 1) // fallback until the panorama streams in
   373→
   374→    scene.backgroundRotation.set(BACKDROP_PITCH_X, ENV_ROTATION_Y, 0)
   375→    scene.environmentRotation.set(BACKDROP_PITCH_X, ENV_ROTATION_Y, 0) // keep reflections aligned
   376→
   377→    const pmrem = new THREE.PMREMGenerator(renderer)
   378→    let lakeTex: THREE.Texture | null = null
   379→    let lakeEnvRT: THREE.WebGLRenderTarget | null = null
   380→
   381→    new THREE.TextureLoader().load(
   382→      ENV_URL,
   383→      (tex) => {
   384→        if (disposed) {
   385→          tex.dispose()
   386→          return
   387→        }
   388→        tex.mapping = THREE.EquirectangularReflectionMapping // 2:1 photo → skybox + IBL
   389→        tex.colorSpace = THREE.SRGBColorSpace
   390→        tex.anisotropy = Math.min(8, renderer.capabilities.getMaxAnisotropy())
   391→        lakeTex = tex
   392→        scene.background = tex
   393→        scene.backgroundIntensity = 1.0
   394→        lakeEnvRT = pmrem.fromEquirectangular(tex) // roughness-aware reflections
   395→        scene.environment = lakeEnvRT.texture
   396→      },
   397→      undefined,
   398→      (error) => console.warn('[scroll-experience] panorama failed to load:', error),
   399→    )
   400→
   401→    /* ── Camera ────────────────────────────────────────────────────── */
   402→    const camera = new THREE.PerspectiveCamera(FOV_DESKTOP, window.innerWidth / window.innerHeight, 0.1, 160)
   403→
   404→    /* ── Lights ────────────────────────────────────────────────────── */

   406→    key.position.set(3, 14, 4) // steep angle → tight shadow that hugs the tires
   407→    key.angle = 0.55
   408→    key.penumbra = 0.55
   409→    key.decay = 2
   410→    key.castShadow = true
   411→    key.shadow.mapSize.set(2048, 2048)
   412→    key.shadow.bias = -0.0002
   413→    key.shadow.normalBias = 0.02
   414→    key.shadow.camera.near = 3
   415→    key.shadow.camera.far = 30
   416→    key.target.position.set(0, 0.5, 0)
   417→    scene.add(key, key.target)
   418→
   419→    const rim = new THREE.DirectionalLight(0x9fc0ee, 2.4) // moonlit edge light
   420→    rim.position.set(-8, 5, -6)
   421→    scene.add(rim)
   422→
   423→    scene.add(new THREE.HemisphereLight(0x27364e, 0x0a0c10, 0.5)) // night sky / asphalt bounce
   424→
   425→    /* ── Floor — invisible shadow-catcher ───────────────────────────
   426→     * GROUNDING comes from CAMERA GEOMETRY, not from a visible 3D floor:
   427→     * every keyframe is raised (~1.4–2.2 u) and aimed DOWN at the car so
   428→     * the wheels land BELOW the photo's quay-railing line, on the lot
   429→     * asphalt painted in the panorama. This transparent catcher then adds
   430→     * the real cast shadow that welds the tires to that asphalt.
