   104→function makeSoftCircleTexture(): THREE.CanvasTexture {
   105→  const canvas = document.createElement('canvas')
   106→  canvas.width = canvas.height = 128
   107→  const ctx = canvas.getContext('2d')!
   108→  const g = ctx.createRadialGradient(64, 64, 0, 64, 64, 64)

   110→  g.addColorStop(0.25, 'rgba(255,255,255,0.45)')
   111→  g.addColorStop(0.6, 'rgba(255,255,255,0.14)')
   112→  g.addColorStop(1, 'rgba(255,255,255,0)')

   114→  ctx.fillRect(0, 0, 128, 128)
   115→  const tex = new THREE.CanvasTexture(canvas)
   116→  tex.colorSpace = THREE.SRGBColorSpace

   151→type CarRig = {
   152→  car: THREE.Group
   153→  anchorLeft: THREE.Object3D // rear tire contact patches (smoke emitters)
   154→  anchorRight: THREE.Object3D
   155→}
   156→
   157→/**
   158→ * Normalizes ANY car glTF so the camera keyframes above fit without
   159→ * retuning: centers the footprint, grounds it at y = 0, scales the longest
   160→ * horizontal span to TARGET_LENGTH, repaints body/glass, hides showroom
   161→ * plates and enables shadows.
   162→ */
   163→function buildCarRig(source: THREE.Object3D): CarRig {
   164→  const model = source.clone(true)
   165→
   166→  const box = new THREE.Box3().setFromObject(model)
   167→  const size = box.getSize(new THREE.Vector3())
   168→  const center = box.getCenter(new THREE.Vector3())
   169→  const scale = TARGET_LENGTH / Math.max(size.x, size.z)
   170→  model.scale.setScalar(scale)
   171→  model.position.set(-center.x * scale, -box.min.y * scale, -center.z * scale)
   172→  if (FLIP_MODEL) model.rotation.y = Math.PI
   173→
   174→  const gray = satinGrayPaint()
   175→  const glass = darkGlass()
   176→  const junk: THREE.Object3D[] = []
   177→  model.traverse((obj) => {
   178→    if (!(obj instanceof THREE.Mesh)) return
   179→    obj.castShadow = true
   180→    obj.receiveShadow = false
   181→    const mats = Array.isArray(obj.material) ? [...obj.material] : [obj.material]
   182→    let replaced = false
   183→    for (let i = 0; i < mats.length; i++) {
   184→      const name = mats[i]?.name ?? ''
   185→      if (BODY_PAINT.has(name)) {
   186→        mats[i] = gray
   187→        replaced = true
   188→      } else if (GLASS.has(name)) {
   189→        mats[i] = glass
   190→        replaced = true
   191→      } else if (mats[i] && (mats[i] as THREE.MeshStandardMaterial).isMeshStandardMaterial) {
   192→        ;(mats[i] as THREE.MeshStandardMaterial).envMapIntensity = 0.7
   193→      }
   194→    }
   195→    if (replaced) obj.material = Array.isArray(obj.material) ? mats : mats[0]
   196→    // hide giant flat plates (Sketchfab showroom floors)
   197→    const b = new THREE.Box3().setFromObject(obj)
   198→    if (b.max.y - b.min.y < 0.12 && b.max.x - b.min.x > 3 && b.max.z - b.min.z > 3) junk.push(obj)
   199→  })
   200→  junk.forEach((m) => (m.visible = false))
   201→
   202→  const car = new THREE.Group()
   203→  car.add(model)
   204→
   205→  // Smoke anchors at the rear tire contact patches (rear = −X)
   206→  const anchorLeft = new THREE.Object3D()
   207→  anchorLeft.position.set(-TARGET_LENGTH * 0.3, 0.16, TARGET_LENGTH * 0.155)
   208→  const anchorRight = new THREE.Object3D()
   209→  anchorRight.position.set(-TARGET_LENGTH * 0.3, 0.16, -TARGET_LENGTH * 0.155)
   210→  car.add(anchorLeft, anchorRight)
   211→
   212→  return { car, anchorLeft, anchorRight }
   213→}

   216→ * Tire smoke — pooled CPU particles + point-sprite shader

   219→const SMOKE_VERTEX = /* glsl */ `
   220→  attribute float aSize;
   221→  attribute float aAlpha;
   222→  uniform float uScale;
   223→  varying float vAlpha;
   224→  void main() {
   225→    vec4 mv = modelViewMatrix * vec4(position, 1.0);
   226→    float dist = max(0.1, -mv.z);
   227→    // cap the sprite size and fade puffs that get right next to the
   228→    // camera (close-up states park the lens near the rear wheels)
   229→    gl_PointSize = min(aSize * uScale / dist, 190.0);
   230→    vAlpha = aAlpha * smoothstep(0.7, 2.2, dist);
   231→    gl_Position = projectionMatrix * mv;
   232→  }
   233→`
   234→
   235→const SMOKE_FRAGMENT = /* glsl */ `
   236→  uniform sampler2D uMap;
   237→  varying float vAlpha;
   238→  void main() {
   239→    float a = texture2D(uMap, gl_PointCoord).a * vAlpha * 0.7; // night-subtle
   240→    if (a < 0.004) discard;
   241→    gl_FragColor = vec4(0.84, 0.86, 0.9, a);
   242→  }
   243→`

   245→type SmokeParticle = {
   246→  life: number
   247→  maxLife: number
   248→  x: number
   249→  y: number
   250→  z: number
   251→  vx: number
   252→  vy: number
   253→  vz: number
   254→  size: number
   255→}

   257→function createSmokeSystem(softTex: THREE.Texture) {
   258→  const particles: SmokeParticle[] = Array.from({ length: SMOKE_COUNT }, () => ({
   259→    life: 0,
   260→    maxLife: 1,
   261→    x: 0,
   262→    y: -50,
   263→    z: 0,
   264→    vx: 0,
   265→    vy: 0,
   266→    vz: 0,
   267→    size: 1,
   268→  }))
   269→  const position = new Float32Array(SMOKE_COUNT * 3)
   270→  const aSize = new Float32Array(SMOKE_COUNT)
   271→  const aAlpha = new Float32Array(SMOKE_COUNT)
   272→  const geometry = new THREE.BufferGeometry()
   273→  geometry.setAttribute('position', new THREE.BufferAttribute(position, 3))
   274→  geometry.setAttribute('aSize', new THREE.BufferAttribute(aSize, 1))
   275→  geometry.setAttribute('aAlpha', new THREE.BufferAttribute(aAlpha, 1))
   276→
   277→  const material = new THREE.ShaderMaterial({
   278→    uniforms: { uScale: { value: 600 }, uMap: { value: softTex } },
   279→    vertexShader: SMOKE_VERTEX,
   280→    fragmentShader: SMOKE_FRAGMENT,
   281→    transparent: true,
   282→    depthWrite: false,
   283→  })
   284→
   285→  const points = new THREE.Points(geometry, material)
   286→  points.frustumCulled = false
   287→  points.renderOrder = 5
   288→
   289→  const state = { cursor: 0, spawnAcc: 0 }
   290→
   291→  /** dt seconds · emitters are the two rear-tire anchor world positions */
   292→  function update(dt: number, left: THREE.Vector3, right: THREE.Vector3, spawn: boolean) {
   293→    state.spawnAcc += dt * (spawn ? SMOKE_PER_SECOND : 0)
   294→    while (state.spawnAcc >= 1) {
   295→      state.spawnAcc -= 1
   296→      state.cursor = (state.cursor + 1) % SMOKE_COUNT
   297→      const p = particles[state.cursor]
   298→      const src = state.cursor % 2 === 0 ? left : right
   299→      p.life = 0.0001
   300→      p.maxLife = 1.8 + Math.random() * 1.0
   301→      p.x = src.x + (Math.random() - 0.5) * 0.24
   302→      p.y = 0.12 + Math.random() * 0.08
   303→      p.z = src.z + (Math.random() - 0.5) * 0.24
   304→      // drift backwards (−X) and outward away from the centerline
   305→      p.vx = -(0.5 + Math.random() * 0.7)
   306→      p.vz = Math.sign(src.z || 1) * (0.6 + Math.random() * 0.8) + (Math.random() - 0.5) * 0.5
   307→      p.vy = 0.18 + Math.random() * 0.3
   308→      p.size = 0.35 + Math.random() * 0.3
   309→    }
   310→
   311→    for (let i = 0; i < SMOKE_COUNT; i++) {
   312→      const p = particles[i]
   313→      if (p.life > 0) {
   314→        p.life += dt
   315→        if (p.life >= p.maxLife) {
   316→          p.life = 0
   317→        } else {
   318→          const drag = Math.exp(-1.1 * dt)
   319→          p.vx *= drag
   320→          p.vz *= drag
   321→          p.vy = p.vy * Math.exp(-0.7 * dt) + 0.14 * dt
   322→          p.x += p.vx * dt
   323→          p.y += p.vy * dt
   324→          p.z += p.vz * dt
   325→        }
   326→      }
   327→      const t = p.life > 0 ? p.life / p.maxLife : 0
   328→      position[i * 3] = p.x
   329→      position[i * 3 + 1] = p.life > 0 ? p.y : -50
   330→      position[i * 3 + 2] = p.z
   331→      aSize[i] = p.size + t * 2.0
   332→      aAlpha[i] = p.life > 0 ? Math.min(t * 6, 1) * Math.pow(1 - t, 1.1) * 0.4 : 0
   333→    }
   334→    geometry.attributes.position.needsUpdate = true
   335→    geometry.attributes.aSize.needsUpdate = true
   336→    geometry.attributes.aAlpha.needsUpdate = true
   337→  }
   338→
   339→  function dispose() {
   340→    geometry.dispose()
   341→    material.dispose()
   342→  }
   343→
   344→  return { points, material, update, dispose }
   345→}

   351→type FlatKey = { px: number; py: number; pz: number; tx: number; ty: number; tz: number }
   352→
   353→/** Flatten a keyframe; on portrait screens the pos→target offset is
   354→ *  scaled by mobileF so the car never clips out of the narrow viewport. */
   355→function flattenKey(k: CamKey, mobile: boolean): FlatKey {
   356→  let [px, py, pz] = k.pos
   357→  const [tx, ty, tz] = k.target
   358→  if (mobile) {
   359→    px = tx + (px - tx) * k.mobileF
   360→    py = ty + (py - ty) * k.mobileF
   361→    pz = tz + (pz - tz) * k.mobileF * (k.mobileHeadOn ?? 1)
   362→  }
   363→  return { px, py, pz, tx, ty, tz }
   364→}

   405→export default function ScrollExperience() {
   406→  const canvasRef = useRef<HTMLCanvasElement>(null)
   407→  const trackRef = useRef<HTMLDivElement>(null)
   408→  const heroLayerRef = useRef<HTMLDivElement>(null)
   409→  const capFrontRef = useRef<HTMLDivElement>(null)
   410→  const capRearRef = useRef<HTMLDivElement>(null)
   411→  const endCardRef = useRef<HTMLDivElement>(null)
   412→
   413→  useEffect(() => {
   414→    const canvas = canvasRef.current
   415→    const track = trackRef.current
   416→    const heroLayer = heroLayerRef.current
   417→    const capFront = capFrontRef.current
   418→    const capRear = capRearRef.current
   419→    const endCard = endCardRef.current
   420→    if (!canvas || !track || !heroLayer || !capFront || !capRear || !endCard) return
   421→
   422→    let disposed = false
   423→    gsap.registerPlugin(ScrollTrigger)
   424→
   425→    const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches
   426→
   427→    /* ── Renderer ──────────────────────────────────────────────────── */
   428→    const renderer = new THREE.WebGLRenderer({
   429→      canvas,
   430→      antialias: true,
   431→      alpha: false,
   432→      powerPreference: 'high-performance',
   433→    })
   434→    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)) // perf cap
   435→    renderer.setSize(window.innerWidth, window.innerHeight, false)
   436→    renderer.shadowMap.enabled = true
   437→    // three r186 removed PCFSoftShadowMap — PCFShadowMap is the supported
   438→    // filtered shadow path (renders soft contacts with a 2K map)
   439→    renderer.shadowMap.type = THREE.PCFShadowMap
   440→    renderer.toneMapping = THREE.ACESFilmicToneMapping
   441→    renderer.toneMappingExposure = 1.15 // lifts the night IBL without clipping the city lights
   442→    renderer.outputColorSpace = THREE.SRGBColorSpace
   443→
   444→    /* ── Scene: lakeside night city — 360° equirect panorama ─────
   445→     * The SAME texture is the visible backdrop and the IBL source
   446→     * (PMREM pre-filter), so the metallic paint reflects exactly the
   447→     * skyline / lake / bridges you see behind the car. */
   448→    const scene = new THREE.Scene()
   449→    renderer.setClearColor(0x04050a, 1) // fallback until the panorama streams in
   450→
   451→    scene.backgroundRotation.set(BACKDROP_PITCH_X, ENV_ROTATION_Y, 0)
   452→    scene.environmentRotation.set(BACKDROP_PITCH_X, ENV_ROTATION_Y, 0) // keep reflections aligned
   453→
   454→    const pmrem = new THREE.PMREMGenerator(renderer)
   455→    let lakeTex: THREE.Texture | null = null
   456→    let lakeEnvRT: THREE.WebGLRenderTarget | null = null
   457→
   458→    new THREE.TextureLoader().load(
   459→      ENV_URL,
   460→      (tex) => {
   461→        if (disposed) {
   462→          tex.dispose()
   463→          return
   464→        }
   465→        tex.mapping = THREE.EquirectangularReflectionMapping // 2:1 photo → skybox + IBL
   466→        tex.colorSpace = THREE.SRGBColorSpace
   467→        tex.anisotropy = Math.min(8, renderer.capabilities.getMaxAnisotropy())
   468→        lakeTex = tex
   469→        scene.background = tex
   470→        scene.backgroundIntensity = 1.0
   471→        lakeEnvRT = pmrem.fromEquirectangular(tex) // roughness-aware reflections
   472→        scene.environment = lakeEnvRT.texture
   473→      },
   474→      undefined,
   475→      (error) => console.warn('[scroll-experience] panorama failed to load:', error),
   476→    )
   477→
   478→    /* ── Camera ────────────────────────────────────────────────────── */
   479→    const camera = new THREE.PerspectiveCamera(FOV_DESKTOP, window.innerWidth / window.innerHeight, 0.1, 160)
