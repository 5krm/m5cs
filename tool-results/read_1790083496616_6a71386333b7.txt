   214→type CarRig = {
   215→  car: THREE.Group
   216→}
   217→
   218→/**
   219→ * Normalizes ANY car glTF so the camera keyframes above fit without
   220→ * retuning: centers the footprint, grounds it at y = 0, scales the longest
   221→ * horizontal span to TARGET_LENGTH, repaints body/glass, hides showroom
   222→ * plates and enables shadows.
   223→ */
   224→function buildCarRig(source: THREE.Object3D): CarRig {
   225→  const model = source.clone(true)
   226→
   227→  const box = new THREE.Box3().setFromObject(model)
   228→  const size = box.getSize(new THREE.Vector3())
   229→  const center = box.getCenter(new THREE.Vector3())
   230→  const scale = TARGET_LENGTH / Math.max(size.x, size.z)
   231→  model.scale.setScalar(scale)
   232→  model.position.set(-center.x * scale, -box.min.y * scale, -center.z * scale)
   233→  if (FLIP_MODEL) model.rotation.y = Math.PI
   234→
   235→  const gray = satinGrayPaint()
   236→  const glass = darkGlass()
   237→  const junk: THREE.Object3D[] = []
   238→  model.traverse((obj) => {
   239→    if (!(obj instanceof THREE.Mesh)) return
   240→    obj.castShadow = true
   241→    obj.receiveShadow = false
   242→    const mats = Array.isArray(obj.material) ? [...obj.material] : [obj.material]
   243→    let replaced = false
   244→    for (let i = 0; i < mats.length; i++) {
   245→      const name = mats[i]?.name ?? ''
   246→      if (BODY_PAINT.has(name)) {
   247→        mats[i] = gray
   248→        replaced = true
   249→      } else if (GLASS.has(name)) {
   250→        mats[i] = glass
   251→        replaced = true
   252→      } else if (mats[i] && (mats[i] as THREE.MeshStandardMaterial).isMeshStandardMaterial) {
   253→        ;(mats[i] as THREE.MeshStandardMaterial).envMapIntensity = 0.7
   254→      }
   255→    }
   256→    if (replaced) obj.material = Array.isArray(obj.material) ? mats : mats[0]
   257→    // hide giant flat plates (Sketchfab showroom floors)
   258→    const b = new THREE.Box3().setFromObject(obj)
   259→    if (b.max.y - b.min.y < 0.12 && b.max.x - b.min.x > 3 && b.max.z - b.min.z > 3) junk.push(obj)
   260→  })
   261→  junk.forEach((m) => (m.visible = false))
   262→
   263→  const car = new THREE.Group()
   264→  car.add(model)
   265→
   266→  return { car }
   267→}

   273→type FlatKey = { px: number; py: number; pz: number; tx: number; ty: number; tz: number }
   274→
   275→/** Flatten a keyframe; on portrait screens the pos→target offset is
   276→ *  scaled by mobileF so the car never clips out of the narrow viewport. */
   277→function flattenKey(k: CamKey, mobile: boolean): FlatKey {
   278→  let [px, py, pz] = k.pos
   279→  const [tx, ty, tz] = k.target
   280→  if (mobile) {
   281→    px = tx + (px - tx) * k.mobileF
   282→    py = ty + (py - ty) * k.mobileF
   283→    pz = tz + (pz - tz) * k.mobileF * (k.mobileHeadOn ?? 1)
   284→  }
   285→  return { px, py, pz, tx, ty, tz }
   286→}
