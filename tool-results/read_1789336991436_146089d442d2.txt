   151→   151→type CarRig = {
   152→   152→  car: THREE.Group
   153→   153→  anchorLeft: THREE.Object3D // rear tire contact patches (smoke emitters)
   154→   154→  anchorRight: THREE.Object3D

   157→   157→/**
   158→   158→ * Normalizes ANY car glTF so the camera keyframes above fit without
   159→   159→ * retuning: centers the footprint, grounds it at y = 0, scales the longest
   160→   160→ * horizontal span to TARGET_LENGTH, repaints body/glass, hides showroom
   161→   161→ * plates and enables shadows.
   162→   162→ */
   163→   163→function buildCarRig(source: THREE.Object3D): CarRig {
   164→   164→  const model = source.clone(true)
   165→   165→
   166→   166→  const box = new THREE.Box3().setFromObject(model)
   167→   167→  const size = box.getSize(new THREE.Vector3())
   168→   168→  const center = box.getCenter(new THREE.Vector3())
   169→   169→  const scale = TARGET_LENGTH / Math.max(size.x, size.z)
   170→   170→  model.scale.setScalar(scale)
   171→   171→  model.position.set(-center.x * scale, -box.min.y * scale, -center.z * scale)
   172→   172→  if (FLIP_MODEL) model.rotation.y = Math.PI
   173→   173→
   174→   174→  const gray = satinGrayPaint()
   175→   175→  const glass = darkGlass()
   176→   176→  const junk: THREE.Object3D[] = []
   177→   177→  model.traverse((obj) => {
   178→   178→    if (!(obj instanceof THREE.Mesh)) return
   179→   179→    obj.castShadow = true
   180→   180→    obj.receiveShadow = false
   181→   181→    const mats = Array.isArray(obj.material) ? [...obj.material] : [obj.material]
   182→   182→    let replaced = false
   183→   183→    for (let i = 0; i < mats.length; i++) {
   184→   184→      const name = mats[i]?.name ?? ''
   185→   185→      if (BODY_PAINT.has(name)) {
   186→   186→        mats[i] = gray
   187→   187→        replaced = true
   188→   188→      } else if (GLASS.has(name)) {
   189→   189→        mats[i] = glass
   190→   190→        replaced = true
   191→   191→      } else if (mats[i] && (mats[i] as THREE.MeshStandardMaterial).isMeshStandardMaterial) {
   192→   192→        ;(mats[i] as THREE.MeshStandardMaterial).envMapIntensity = 0.7
   193→   193→      }
   194→   194→    }
   195→   195→    if (replaced) obj.material = Array.isArray(obj.material) ? mats : mats[0]
   196→   196→    // hide giant flat plates (Sketchfab showroom floors)
   197→   197→    const b = new THREE.Box3().setFromObject(obj)
   198→   198→    if (b.max.y - b.min.y < 0.12 && b.max.x - b.min.x > 3 && b.max.z - b.min.z > 3) junk.push(obj)
   199→   199→  })
   200→   200→  junk.forEach((m) => (m.visible = false))
   201→   201→
   202→   202→  const car = new THREE.Group()
   203→   203→  car.add(model)
   204→   204→
   205→   205→  // Smoke anchors at the rear tire contact patches (rear = −X)
   206→   206→  const anchorLeft = new THREE.Object3D()
   207→   207→  anchorLeft.position.set(-TARGET_LENGTH * 0.3, 0.16, TARGET_LENGTH * 0.155)
   208→   208→  const anchorRight = new THREE.Object3D()
   209→   209→  anchorRight.position.set(-TARGET_LENGTH * 0.3, 0.16, -TARGET_LENGTH * 0.155)
   210→   210→  car.add(anchorLeft, anchorRight)
   211→   211→
   212→   212→  return { car, anchorLeft, anchorRight }
   213→   213→}

   219→   219→const SMOKE_VERTEX = /* glsl */ `
   220→   220→  attribute float aSize;
   221→   221→  attribute float aAlpha;
   222→   222→  uniform float uScale;
   223→   223→  varying float vAlpha;
   224→   224→  void main() {
   225→   225→    vec4 mv = modelViewMatrix * vec4(position, 1.0);
   226→   226→    float dist = max(0.1, -mv.z);
   227→   227→    // cap the sprite size and fade puffs that get right next to the
   228→   228→    // camera (close-up states park the lens near the rear wheels)
   229→   229→    gl_PointSize = min(aSize * uScale / dist, 190.0);
   230→   230→    vAlpha = aAlpha * smoothstep(0.7, 2.2, dist);
   231→   231→    gl_Position = projectionMatrix * mv;
   232→   232→  }
   233→   233→`
   234→   234→
   235→   235→const SMOKE_FRAGMENT = /* glsl */ `
   236→   236→  uniform sampler2D uMap;
   237→   237→  varying float vAlpha;
   238→   238→  void main() {
   239→   239→    float a = texture2D(uMap, gl_PointCoord).a * vAlpha * 0.7; // night-subtle
   240→   240→    if (a < 0.004) discard;
   241→   241→    gl_FragColor = vec4(0.84, 0.86, 0.9, a);
   242→   242→  }
   243→   243→`

   257→   257→function createSmokeSystem(softTex: THREE.Texture) {
   258→   258→  const particles: SmokeParticle[] = Array.from({ length: SMOKE_COUNT }, () => ({
   259→   259→    life: 0,
   260→   260→    maxLife: 1,
   261→   261→    x: 0,
   262→   262→    y: -50,
   263→   263→    z: 0,
   264→   264→    vx: 0,
   265→   265→    vy: 0,
   266→   266→    vz: 0,
   267→   267→    size: 1,
   268→   268→  }))
   269→   269→  const position = new Float32Array(SMOKE_COUNT * 3)
   270→   270→  const aSize = new Float32Array(SMOKE_COUNT)
   271→   271→  const aAlpha = new Float32Array(SMOKE_COUNT)
   272→   272→  const geometry = new THREE.BufferGeometry()
   273→   273→  geometry.setAttribute('position', new THREE.BufferAttribute(position, 3))
   274→   274→  geometry.setAttribute('aSize', new THREE.BufferAttribute(aSize, 1))
   275→   275→  geometry.setAttribute('aAlpha', new THREE.BufferAttribute(aAlpha, 1))
   276→   276→
   277→   277→  const material = new THREE.ShaderMaterial({
   278→   278→    uniforms: { uScale: { value: 600 }, uMap: { value: softTex } },
   279→   279→    vertexShader: SMOKE_VERTEX,
   280→   280→    fragmentShader: SMOKE_FRAGMENT,
   281→   281→    transparent: true,
   282→   282→    depthWrite: false,
   283→   283→  })
   284→   284→
   285→   285→  const points = new THREE.Points(geometry, material)
   286→   286→  points.frustumCulled = false
   287→   287→  points.renderOrder = 5
   288→   288→
   289→   289→  const state = { cursor: 0, spawnAcc: 0 }
   290→   290→
   291→   291→  /** dt seconds · emitters are the two rear-tire anchor world positions */
   292→   292→  function update(dt: number, left: THREE.Vector3, right: THREE.Vector3, spawn: boolean) {
   293→   293→    state.spawnAcc += dt * (spawn ? SMOKE_PER_SECOND : 0)
   294→   294→    while (state.spawnAcc >= 1) {
   295→   295→      state.spawnAcc -= 1
   296→   296→      state.cursor = (state.cursor + 1) % SMOKE_COUNT
   297→   297→      const p = particles[state.cursor]
   298→   298→      const src = state.cursor % 2 === 0 ? left : right
   299→   299→      p.life = 0.0001
   300→   300→      p.maxLife = 1.8 + Math.random() * 1.0
   301→   301→      p.x = src.x + (Math.random() - 0.5) * 0.24
   302→   302→      p.y = 0.12 + Math.random() * 0.08
   303→   303→      p.z = src.z + (Math.random() - 0.5) * 0.24
   304→   304→      // drift backwards (−X) and outward away from the centerline
   305→   305→      p.vx = -(0.5 + Math.random() * 0.7)
   306→   306→      p.vz = Math.sign(src.z || 1) * (0.6 + Math.random() * 0.8) + (Math.random() - 0.5) * 0.5
   307→   307→      p.vy = 0.18 + Math.random() * 0.3
   308→   308→      p.size = 0.35 + Math.random() * 0.3
   309→   309→    }
   310→   310→
   311→   311→    for (let i = 0; i < SMOKE_COUNT; i++) {
   312→   312→      const p = particles[i]
   313→   313→      if (p.life > 0) {
   314→   314→        p.life += dt
   315→   315→        if (p.life >= p.maxLife) {
   316→   316→          p.life = 0
   317→   317→        } else {
   318→   318→          const drag = Math.exp(-1.1 * dt)
   319→   319→          p.vx *= drag
   320→   320→          p.vz *= drag
   321→   321→          p.vy = p.vy * Math.exp(-0.7 * dt) + 0.14 * dt
   322→   322→          p.x += p.vx * dt
   323→   323→          p.y += p.vy * dt
   324→   324→          p.z += p.vz * dt
   325→   325→        }
   326→   326→      }
   327→   327→      const t = p.life > 0 ? p.life / p.maxLife : 0
   328→   328→      position[i * 3] = p.x
   329→   329→      position[i * 3 + 1] = p.life > 0 ? p.y : -50
   330→   330→      position[i * 3 + 2] = p.z
   331→   331→      aSize[i] = p.size + t * 2.0
   332→   332→      aAlpha[i] = p.life > 0 ? Math.min(t * 6, 1) * Math.pow(1 - t, 1.1) * 0.4 : 0
   333→   333→    }
   334→   334→    geometry.attributes.position.needsUpdate = true
   335→   335→    geometry.attributes.aSize.needsUpdate = true
   336→   336→    geometry.attributes.aAlpha.needsUpdate = true
   337→   337→  }
   338→   338→
   339→   339→  function dispose() {
   340→   340→    geometry.dispose()
   341→   341→    material.dispose()
   342→   342→  }
   343→   343→
   344→   344→  return { points, material, update, dispose }
   345→   345→}

   353→   353→/** Flatten a keyframe; on portrait screens the pos→target offset is
   354→   354→ *  scaled by mobileF so the car never clips out of the narrow viewport. */
   355→   355→function flattenKey(k: CamKey, mobile: boolean): FlatKey {
   356→   356→  let [px, py, pz] = k.pos
   357→   357→  const [tx, ty, tz] = k.target
   358→   358→  if (mobile) {
   359→   359→    px = tx + (px - tx) * k.mobileF
   360→   360→    py = ty + (py - ty) * k.mobileF
   361→   361→    pz = tz + (pz - tz) * k.mobileF * (k.mobileHeadOn ?? 1)
   362→   362→  }
   363→   363→  return { px, py, pz, tx, ty, tz }
   364→   364→}
