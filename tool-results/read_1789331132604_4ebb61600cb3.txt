   164→   164→type CarRig = {
   165→   165→  car: THREE.Group
   166→   166→  anchorLeft: THREE.Object3D // rear tire contact patches (smoke emitters)
   167→   167→  anchorRight: THREE.Object3D

   170→   170→/**
   171→   171→ * Normalizes ANY car glTF so the camera keyframes above fit without
   172→   172→ * retuning: centers the footprint, grounds it at y = 0, scales the longest
   173→   173→ * horizontal span to TARGET_LENGTH, repaints body/glass, hides showroom
   174→   174→ * plates and enables shadows.
   175→   175→ */
   176→   176→function buildCarRig(source: THREE.Object3D): CarRig {
   177→   177→  const model = source.clone(true)
   178→   178→
   179→   179→  const box = new THREE.Box3().setFromObject(model)
   180→   180→  const size = box.getSize(new THREE.Vector3())
   181→   181→  const center = box.getCenter(new THREE.Vector3())
   182→   182→  const scale = TARGET_LENGTH / Math.max(size.x, size.z)
   183→   183→  model.scale.setScalar(scale)
   184→   184→  model.position.set(-center.x * scale, -box.min.y * scale, -center.z * scale)
   185→   185→  if (FLIP_MODEL) model.rotation.y = Math.PI
   186→   186→
   187→   187→  const gray = satinGrayPaint()
   188→   188→  const glass = darkGlass()
   189→   189→  const junk: THREE.Object3D[] = []
   190→   190→  model.traverse((obj) => {
   191→   191→    if (!(obj instanceof THREE.Mesh)) return
   192→   192→    obj.castShadow = true
   193→   193→    obj.receiveShadow = false
   194→   194→    const mats = Array.isArray(obj.material) ? [...obj.material] : [obj.material]
   195→   195→    let replaced = false
   196→   196→    for (let i = 0; i < mats.length; i++) {
   197→   197→      const name = mats[i]?.name ?? ''
   198→   198→      if (BODY_PAINT.has(name)) {
   199→   199→        mats[i] = gray
   200→   200→        replaced = true
   201→   201→      } else if (GLASS.has(name)) {
   202→   202→        mats[i] = glass
   203→   203→        replaced = true
   204→   204→      } else if (mats[i] && (mats[i] as THREE.MeshStandardMaterial).isMeshStandardMaterial) {
   205→   205→        ;(mats[i] as THREE.MeshStandardMaterial).envMapIntensity = 0.7
   206→   206→      }
   207→   207→    }
   208→   208→    if (replaced) obj.material = Array.isArray(obj.material) ? mats : mats[0]
   209→   209→    // hide giant flat plates (Sketchfab showroom floors)
   210→   210→    const b = new THREE.Box3().setFromObject(obj)
   211→   211→    if (b.max.y - b.min.y < 0.12 && b.max.x - b.min.x > 3 && b.max.z - b.min.z > 3) junk.push(obj)
   212→   212→  })
   213→   213→  junk.forEach((m) => (m.visible = false))
   214→   214→
   215→   215→  const car = new THREE.Group()
   216→   216→  car.add(model)
   217→   217→
   218→   218→  // Smoke anchors at the rear tire contact patches (rear = −X)
   219→   219→  const anchorLeft = new THREE.Object3D()
   220→   220→  anchorLeft.position.set(-TARGET_LENGTH * 0.3, 0.16, TARGET_LENGTH * 0.155)
   221→   221→  const anchorRight = new THREE.Object3D()
   222→   222→  anchorRight.position.set(-TARGET_LENGTH * 0.3, 0.16, -TARGET_LENGTH * 0.155)
   223→   223→  car.add(anchorLeft, anchorRight)
   224→   224→
   225→   225→  return { car, anchorLeft, anchorRight }
   226→   226→}

   232→   232→const SMOKE_VERTEX = /* glsl */ `
   233→   233→  attribute float aSize;
   234→   234→  attribute float aAlpha;
   235→   235→  uniform float uScale;
   236→   236→  varying float vAlpha;
   237→   237→  void main() {
   238→   238→    vec4 mv = modelViewMatrix * vec4(position, 1.0);
   239→   239→    float dist = max(0.1, -mv.z);
   240→   240→    // cap the sprite size and fade puffs that get right next to the
   241→   241→    // camera (close-up states park the lens near the rear wheels)
   242→   242→    gl_PointSize = min(aSize * uScale / dist, 300.0);
   243→   243→    vAlpha = aAlpha * smoothstep(0.7, 2.2, dist);
   244→   244→    gl_Position = projectionMatrix * mv;
   245→   245→  }
   246→   246→`
   247→   247→
   248→   248→const SMOKE_FRAGMENT = /* glsl */ `
   249→   249→  uniform sampler2D uMap;
   250→   250→  varying float vAlpha;
   251→   251→  void main() {
   252→   252→    float a = texture2D(uMap, gl_PointCoord).a * vAlpha;
   253→   253→    if (a < 0.004) discard;
   254→   254→    gl_FragColor = vec4(0.84, 0.86, 0.9, a);
   255→   255→  }
   256→   256→`

   270→   270→function createSmokeSystem(softTex: THREE.Texture) {
   271→   271→  const particles: SmokeParticle[] = Array.from({ length: SMOKE_COUNT }, () => ({
   272→   272→    life: 0,
   273→   273→    maxLife: 1,
   274→   274→    x: 0,
   275→   275→    y: -50,
   276→   276→    z: 0,
   277→   277→    vx: 0,
   278→   278→    vy: 0,
   279→   279→    vz: 0,
   280→   280→    size: 1,
   281→   281→  }))
   282→   282→  const position = new Float32Array(SMOKE_COUNT * 3)
   283→   283→  const aSize = new Float32Array(SMOKE_COUNT)
   284→   284→  const aAlpha = new Float32Array(SMOKE_COUNT)
   285→   285→  const geometry = new THREE.BufferGeometry()
   286→   286→  geometry.setAttribute('position', new THREE.BufferAttribute(position, 3))
   287→   287→  geometry.setAttribute('aSize', new THREE.BufferAttribute(aSize, 1))
   288→   288→  geometry.setAttribute('aAlpha', new THREE.BufferAttribute(aAlpha, 1))
   289→   289→
   290→   290→  const material = new THREE.ShaderMaterial({
   291→   291→    uniforms: { uScale: { value: 600 }, uMap: { value: softTex } },
   292→   292→    vertexShader: SMOKE_VERTEX,
   293→   293→    fragmentShader: SMOKE_FRAGMENT,
   294→   294→    transparent: true,
   295→   295→    depthWrite: false,
   296→   296→  })
   297→   297→
   298→   298→  const points = new THREE.Points(geometry, material)
   299→   299→  points.frustumCulled = false
   300→   300→  points.renderOrder = 5
   301→   301→
   302→   302→  const state = { cursor: 0, spawnAcc: 0 }
   303→   303→
   304→   304→  /** dt seconds · emitters are the two rear-tire anchor world positions */
   305→   305→  function update(dt: number, left: THREE.Vector3, right: THREE.Vector3, spawn: boolean) {
   306→   306→    state.spawnAcc += dt * (spawn ? SMOKE_PER_SECOND : 0)
   307→   307→    while (state.spawnAcc >= 1) {
   308→   308→      state.spawnAcc -= 1
   309→   309→      state.cursor = (state.cursor + 1) % SMOKE_COUNT
   310→   310→      const p = particles[state.cursor]
   311→   311→      const src = state.cursor % 2 === 0 ? left : right
   312→   312→      p.life = 0.0001
   313→   313→      p.maxLife = 1.8 + Math.random() * 1.0
   314→   314→      p.x = src.x + (Math.random() - 0.5) * 0.24
   315→   315→      p.y = 0.12 + Math.random() * 0.08
   316→   316→      p.z = src.z + (Math.random() - 0.5) * 0.24
   317→   317→      // drift backwards (−X) and outward away from the centerline
   318→   318→      p.vx = -(0.5 + Math.random() * 0.7)
   319→   319→      p.vz = Math.sign(src.z || 1) * (0.6 + Math.random() * 0.8) + (Math.random() - 0.5) * 0.5
   320→   320→      p.vy = 0.18 + Math.random() * 0.3
   321→   321→      p.size = 0.5 + Math.random() * 0.4
   322→   322→    }
   323→   323→
   324→   324→    for (let i = 0; i < SMOKE_COUNT; i++) {
   325→   325→      const p = particles[i]
   326→   326→      if (p.life > 0) {
   327→   327→        p.life += dt
   328→   328→        if (p.life >= p.maxLife) {
   329→   329→          p.life = 0
   330→   330→        } else {
   331→   331→          const drag = Math.exp(-1.1 * dt)
   332→   332→          p.vx *= drag
   333→   333→          p.vz *= drag
   334→   334→          p.vy = p.vy * Math.exp(-0.7 * dt) + 0.14 * dt
   335→   335→          p.x += p.vx * dt
   336→   336→          p.y += p.vy * dt
   337→   337→          p.z += p.vz * dt
   338→   338→        }
   339→   339→      }
   340→   340→      const t = p.life > 0 ? p.life / p.maxLife : 0
   341→   341→      position[i * 3] = p.x
   342→   342→      position[i * 3 + 1] = p.life > 0 ? p.y : -50
   343→   343→      position[i * 3 + 2] = p.z
   344→   344→      aSize[i] = p.size + t * 2.8
   345→   345→      aAlpha[i] = p.life > 0 ? Math.min(t * 6, 1) * Math.pow(1 - t, 1.1) * 0.4 : 0
   346→   346→    }
   347→   347→    geometry.attributes.position.needsUpdate = true
   348→   348→    geometry.attributes.aSize.needsUpdate = true
   349→   349→    geometry.attributes.aAlpha.needsUpdate = true
   350→   350→  }
   351→   351→
   352→   352→  function dispose() {
   353→   353→    geometry.dispose()
   354→   354→    material.dispose()
   355→   355→  }
   356→   356→
   357→   357→  return { points, material, update, dispose }
   358→   358→}

   366→   366→/** Flatten a keyframe; on portrait screens the pos→target offset is
   367→   367→ *  scaled by mobileF so the car never clips out of the narrow viewport. */
   368→   368→function flattenKey(k: CamKey, mobile: boolean): FlatKey {
   369→   369→  let [px, py, pz] = k.pos
   370→   370→  const [tx, ty, tz] = k.target
   371→   371→  if (mobile) {
   372→   372→    px = tx + (px - tx) * k.mobileF
   373→   373→    py = ty + (py - ty) * k.mobileF
   374→   374→    pz = tz + (pz - tz) * k.mobileF * (k.mobileHeadOn ?? 1)
   375→   375→  }
   376→   376→  return { px, py, pz, tx, ty, tz }
   377→   377→}
